```python
from __future__ import annotations
from pathlib import Path

CONTENT_DIR = "База знаний"


def repo_root(start: Path | None = None) -> Path:
    """Поднимается вверх до папки, содержащей каталог 'База знаний'."""
    p = (start or Path(__file__)).resolve()
    for parent in [p, *p.parents]:
        if (parent / CONTENT_DIR).is_dir():
            return parent
    raise RuntimeError(f"Не найдена папка '{CONTENT_DIR}'")


def content_files(root: Path) -> list[Path]:
    """Все .md-файлы базы знаний, отсортированные по пути."""
    return sorted((root / CONTENT_DIR).rglob("*.md"))


def split_frontmatter(text: str) -> tuple[str | None, str]:
    """Делит текст на (YAML-frontmatter | None, тело).

    Frontmatter — блок между строками '---' в самом начале файла.
    """
    if not text.startswith("---"):
        return None, text
    end = text.find("\n---", 3)
    if end == -1:
        return None, text
    yaml_text = text[3:end].strip("\n")
    body = text[end + 4:]
    return yaml_text, body
```

`scripts/kb_tools/check_frontmatter.py`:

```python
from __future__ import annotations
import sys
from pathlib import Path

import yaml

from kb_tools.common import repo_root, content_files, split_frontmatter

REQUIRED = ["title", "type", "status", "department", "owner",
            "version", "created", "updated", "tags"]
TYPES = {"справочник", "процедура", "регламент", "скрипт",
         "FAQ", "обучение", "стратегия"}
STATUSES = {"черновик", "актуально", "требует обновления", "устарело"}


def load_tags(root: Path) -> set[str]:
    """Читает словарь тегов из scripts/kb_tools/tags.yml."""
    tags_file = root / "scripts" / "kb_tools" / "tags.yml"
    data = yaml.safe_load(tags_file.read_text(encoding="utf-8"))
    return set(data.get("tags", []))


def check_file(path: Path, allowed_tags: set[str]) -> list[str]:
    """Возвращает список ошибок frontmatter для одного файла."""
    text = path.read_text(encoding="utf-8")
    fm, _ = split_frontmatter(text)
    if fm is None:
        return ["нет YAML-frontmatter"]
    try:
        data = yaml.safe_load(fm) or {}
    except yaml.YAMLError as exc:
        return [f"некорректный YAML: {exc}"]

    errors: list[str] = []
    for field in REQUIRED:
        if field not in data or data[field] in (None, "", []):
            errors.append(f"отсутствует обязательное поле '{field}'")
    if data.get("type") and data["type"] not in TYPES:
        errors.append(f"недопустимое значение type: {data['type']}")
    if data.get("status") and data["status"] not in STATUSES:
        errors.append(f"недопустимое значение status: {data['status']}")
    tags = data.get("tags") or []
    if isinstance(tags, list):
        for tag in tags:
            if tag not in allowed_tags:
                errors.append(f"тег вне словаря: {tag}")
    return errors


def main() -> int:
    root = repo_root()
    allowed = load_tags(root)
    total = 0
    for path in content_files(root):
        if path.name == "README.md":
            continue  # README — навигационные, полный frontmatter не требуется
        errors = check_file(path, allowed)
        for err in errors:
            total += 1
            print(f"{path.relative_to(root)}: {err}")
    if total:
        print(f"\nНайдено проблем frontmatter: {total}")
        return 1
    print("Frontmatter: проблем не найдено.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
from __future__ import annotations
import re
import sys

from kb_tools.common import repo_root, content_files

SUSPECT_CHARS = {
    " ": "неразрывный пробел (U+00A0)",
    "​": "пробел нулевой ширины (U+200B)",
    "‌": "ZWNJ (U+200C)",
    "‍": "ZWJ (U+200D)",
    "﻿": "BOM/ZWNBSP (U+FEFF)",
    "­": "мягкий перенос (U+00AD)",
}
MOJIBAKE_MARKERS = ("Ð", "Ñ", "â€", "Ã", "Â»", "Â«")

CYRILLIC = set("абвгдеёжзийклмнопрстуфхцчшщъыьэюя"
               "АБВГДЕЁЖЗИЙКЛМНОПРСТУФХЦЧШЩЪЫЬЭЮЯ")
LATIN = set("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ")
WORD_RE = re.compile(r"[A-Za-zА-Яа-яЁё]+")


def scan_text(text: str) -> list[str]:
    """Возвращает список находок-артефактов кодировки."""
    findings: list[str] = []
    for ch, name in SUSPECT_CHARS.items():
        if ch in text:
            findings.append(f"подозрительный символ: {name}")
    for marker in MOJIBAKE_MARKERS:
        if marker in text:
            findings.append(f"возможный mojibake: последовательность '{marker}'")
            break
    for match in WORD_RE.finditer(text):
        word = match.group()
        has_cyr = any(c in CYRILLIC for c in word)
        has_lat = any(c in LATIN for c in word)
        if has_cyr and has_lat:
            findings.append(f"смешение алфавитов в слове: {word}")
    return findings


def main() -> int:
    root = repo_root()
    total = 0
    for path in content_files(root):
        text = path.read_text(encoding="utf-8")
        for finding in scan_text(text):
            total += 1
            print(f"{path.relative_to(root)}: {finding}")
    if total:
        print(f"\nНайдено артефактов кодировки: {total}")
        return 1
    print("Кодировка: артефактов не найдено.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

`scripts/kb_tools/check_terminology.py`:

```python
from __future__ import annotations
import sys
from pathlib import Path

import yaml

from kb_tools.common import repo_root, content_files, split_frontmatter


def load_rules(root: Path) -> list[dict]:
    """Читает реестр терминов из scripts/kb_tools/terminology.yml."""
    rules_file = root / "scripts" / "kb_tools" / "terminology.yml"
    data = yaml.safe_load(rules_file.read_text(encoding="utf-8"))
    return data.get("terms", [])


def scan_text(text: str, rules: list[dict]) -> list[str]:
    """Возвращает находки неканоничных написаний."""
    findings: list[str] = []
    lowered = text.lower()
    for rule in rules:
        canonical = rule["canonical"]
        for variant in rule.get("forbidden", []):
            if variant.lower() in lowered:
                findings.append(
                    f"неканоничное написание '{variant}' — заменить на '{canonical}'"
                )
    return findings


def main() -> int:
    root = repo_root()
    rules = load_rules(root)
    total = 0
    for path in content_files(root):
        text = path.read_text(encoding="utf-8")
        _, body = split_frontmatter(text)  # frontmatter (теги) не проверяем
        for finding in scan_text(body, rules):
            total += 1
            print(f"{path.relative_to(root)}: {finding}")
    if total:
        print(f"\nНайдено проблем терминологии: {total}")
        return 1
    print("Терминология: проблем не найдено.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
